    palmerpenguins::penguins,

    na = ""
  )
}

1.4 Базова підготовка даних

Code 
penguins_clean <- penguins_raw |>
select(
  species,
  island,
  bill_length_mm,
  bill_depth_mm,
  flipper_length_mm,
  body_mass_g,
  sex
) |>
  filter(
    !is.na(species),
    !is.na(bill_length_mm),
    !is.na(bill_depth_mm),
    !is.na(flipper_length_mm),
    !is.na(body_mass_g)
  )|>
  mutate(
    body_mass_kg = body_mass_g / 1000,
    species = factor(species),
    island = factor(island),
    sex = factor(sex)
  )
